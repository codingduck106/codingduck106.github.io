/** * Team 2813 Inventory Management System — Apps Script backend * ------------------------------------------------------------- * Deploy as a Web App (Deploy > New deployment > Web app). * - Execute as: Me * - Who has access: Anyone with the link (or "Anyone within [org]" if you're on * a Google Workspace domain and want it locked to team accounts) * * Items are keyed by ItemName (no separate ItemID column). * * ITEM TYPES * ----------- * Every item has an ItemType, fixed the first time it's created, which * controls how it's tracked. This matches the three tabs on the request form: * * - "countable" Discrete whole units (gears, sensors, connectors, standoffs, * motors). Quantity is always rounded to the nearest integer. * Default type if none is supplied, so old submissions keep * working. Unit defaults to "pcs". * - "wire" Spooled/length material (wire, tubing, pneumatic line, belt * stock). Quantity is a decimal length in Unit (e.g. "ft", * "in", "m", "cm"). "checkout" cuts a length off the spool * and errors out if there isn't enough left; "add" is used * when a new spool/reel comes into stock. * - "bulk" Loose consumables where an exact count isn't practical * (zip ties, heat shrink, tape, hardware bags). These are * NOT tracked by a number at all — instead the form sends a * qualitative StockStatus of "out" | "low" | "enough", * stored alongside a numeric code (0/1/2, matching the * form's BULK_CODE) purely so the column sorts sensibly. * There's no LowStockThreshold for bulk items — "out" or * "low" triggers the alert directly. * * The web form (see inventory-request.html) POSTs JSON to the resulting * /exec URL. */ // ====== CONFIG ====== const SHEET_NAME = 'Inventory'; const ALERTS_SHEET_NAME = 'Alerts'; const ALERT_EMAILS = ['jazl.ihsan@gmail.com']; // <-- change me const ITEM_TYPES = ['countable', 'wire', 'bulk']; const DEFAULT_ITEM_TYPE = 'countable'; const DEFAULT_UNITS = { countable: 'pcs', wire: 'ft', bulk: '' }; const BULK_STATUSES = ['out', 'low', 'enough']; const BULK_STATUS_CODE = { out: 0, low: 1, enough: 2 }; // matches the form's BULK_CODE const HEADERS = [ 'ItemName', 'ItemType', 'Quantity', 'StockStatus', 'Unit', 'LowStockThreshold', 'Subteams', 'ImageBase64', 'LastUpdated', 'LastUpdatedBy', 'Notes' ]; // Column indices (1-based, matches HEADERS order) const COL = { ItemName: 1, ItemType: 2, Quantity: 3, StockStatus: 4, Unit: 5, LowStockThreshold: 6, Subteams: 7, ImageBase64: 8, LastUpdated: 9, LastUpdatedBy: 10, Notes: 11 }; // ====== ENTRY POINTS ====== function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); // avoid two simultaneous submissions racing on the same row try { const body = JSON.parse(e.postData.contents); const result = handleAction(body); return jsonResponse({ ok: true, result: result }); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } finally { lock.releaseLock(); } } function doGet(e) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const headers = data.shift(); // Optional query param ?itemName=xxx to fetch a single item (skips ImageBase64 // in bulk listing to keep the payload small; include it for single lookups) const itemName = e.parameter.itemName; if (itemName) { const row = data.find(r => String(r[COL.ItemName - 1]).toLowerCase() === String(itemName).toLowerCase()); if (!row) return jsonResponse({ ok: false, error: 'Item not found' }); return jsonResponse({ ok: true, item: rowToObject(headers, row) }); } const items = data.map(r => { const obj = rowToObject(headers, r); delete obj.ImageBase64; // omit heavy field from bulk listing return obj; }); return jsonResponse({ ok: true, items: items }); } // ====== CORE LOGIC ====== /** * Expected payload shapes from the form: * * Countable / Wire: * { * "action": "add" | "checkout" | "set", * "itemName": "WAGO 221 Lever Nut", * "itemType": "countable" | "wire", // only used/needed on create * "unit": "ft", // wire only; optional, only needed on create * "quantityDelta": 3, // for "add" / "checkout"; decimals ok for wire * "quantity": 40, // for "set" (absolute value) * "lowStockThreshold": 10, // optional, only needed on create * "subteam": "Electrical", * "submittedBy": "Jane Doe", * "notes": "restocked from Amazon order" // optional * } * * Bulk: * { * "action": "set", * "itemName": "Zip Ties (Small)", * "itemType": "bulk", * "stockStatus": "out" | "low" | "enough", * "quantity": 0, // numeric code, informational only * "subteam": "Fab", * "submittedBy": "Jane Doe", * "notes": "..." // optional * } * * Items are matched by ItemName (case-insensitive). Since there's no separate * ID field, submissions must use the exact item name consistently — "WAGO * 221" and "wago 221 lever nut" will create two different rows. * * ItemType is fixed at creation time (first submission for a given ItemName). * Later submissions don't need to (and can't) change it. */ function handleAction(body) { if (!body.itemName) throw new Error('itemName is required'); const sheet = getSheet(); const data = sheet.getDataRange().getValues(); let rowIndex = data.findIndex((r, i) => i > 0 && String(r[COL.ItemName - 1]).toLowerCase() === String(body.itemName).toLowerCase()); // ---- Create new item if it doesn't exist ---- if (rowIndex === -1) { const itemType = normalizeItemType(body.itemType); const unit = (itemType === 'wire') ? ((body.unit && String(body.unit).trim()) || DEFAULT_UNITS.wire) : DEFAULT_UNITS[itemType]; const newRow = [ body.itemName, itemType, 0, // Quantity '', // StockStatus (bulk only) unit, body.lowStockThreshold || 0, '', '', new Date(), body.submittedBy || '', body.notes || '' ]; // 11 values, matching the 11 HEADERS above in order — keep these in sync // any time a column is added or removed. sheet.appendRow(newRow); rowIndex = sheet.getLastRow() - 1; // 0-based index into `data`-style array } const sheetRow = rowIndex + 1; // 1-based row number in the actual sheet const itemType = normalizeItemType(sheet.getRange(sheetRow, COL.ItemType).getValue()); let newQty; let bulkStatus = null; if (itemType === 'bulk') { // ---- Bulk: qualitative status instead of a count ---- bulkStatus = String(body.stockStatus || '').trim().toLowerCase(); if (!BULK_STATUSES.includes(bulkStatus)) { throw new Error(`stockStatus must be one of: ${BULK_STATUSES.join(', ')}`); } newQty = BULK_STATUS_CODE[bulkStatus]; sheet.getRange(sheetRow, COL.Quantity).setValue(newQty); sheet.getRange(sheetRow, COL.StockStatus).setValue(bulkStatus); } else { // ---- Countable / Wire: numeric quantity update ---- const currentQty = Number(sheet.getRange(sheetRow, COL.Quantity).getValue()) || 0; if (body.action === 'add') { newQty = currentQty + Number(body.quantityDelta || 0); } else if (body.action === 'checkout') { const delta = Math.abs(Number(body.quantityDelta || 0)); newQty = currentQty - delta; // Wire (spooled) items can't have more cut off than remains. if (itemType === 'wire' && delta > currentQty) { throw new Error(`Not enough ${sheet.getRange(sheetRow, COL.Unit).getValue()} remaining: ` + `have ${currentQty}, tried to check out ${delta}`); } } else if (body.action === 'set') { newQty = Number(body.quantity); } else { throw new Error(`Unknown action: ${body.action}`); } newQty = Math.max(0, newQty); // Countable items are always whole units; wire may be fractional // (e.g. 12.5 ft of wire remaining). if (itemType === 'countable') { newQty = Math.round(newQty); } sheet.getRange(sheetRow, COL.Quantity).setValue(newQty); } // ---- Subteam jurisdiction (merge, dedupe) ---- if (body.subteam) { const existing = String(sheet.getRange(sheetRow, COL.Subteams).getValue() || ''); const set = new Set(existing.split(',').map(s => s.trim()).filter(Boolean)); set.add(body.subteam.trim()); sheet.getRange(sheetRow, COL.Subteams).setValue(Array.from(set).join(',')); } // ---- Image (optional) ---- if (body.image) { sheet.getRange(sheetRow, COL.ImageBase64).setValue(body.image); } // ---- Metadata ---- sheet.getRange(sheetRow, COL.LastUpdated).setValue(new Date()); if (body.submittedBy) sheet.getRange(sheetRow, COL.LastUpdatedBy).setValue(body.submittedBy); if (body.notes) sheet.getRange(sheetRow, COL.Notes).setValue(body.notes); // ---- Low stock check (type-aware) ---- const itemName = sheet.getRange(sheetRow, COL.ItemName).getValue(); if (itemType === 'bulk') { if (bulkStatus === 'out' || bulkStatus === 'low') { flagLowStock(itemName, `marked ${bulkStatus.toUpperCase()}`); } } else { const threshold = Number(sheet.getRange(sheetRow, COL.LowStockThreshold).getValue()) || 0; if (threshold > 0 && newQty <= threshold) { const unit = sheet.getRange(sheetRow, COL.Unit).getValue(); const unitLabel = unit ? ` ${unit}` : ''; flagLowStock(itemName, `at ${newQty}${unitLabel}, at or below its threshold of ${threshold}${unitLabel}`); } } return { itemName: body.itemName, itemType: itemType, newQuantity: newQty, stockStatus: bulkStatus }; } // ====== HELPERS ====== function normalizeItemType(rawType) { const t = String(rawType || '').trim().toLowerCase(); if (t === 'length') return 'wire'; // backwards-compat with the old type name return ITEM_TYPES.includes(t) ? t : DEFAULT_ITEM_TYPE; } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); } return sheet; } function rowToObject(headers, row) { const obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function flagLowStock(itemName, detailText) { // Log to an Alerts sheet so there's a running history, and email the team. const ss = SpreadsheetApp.getActiveSpreadsheet(); let alerts = ss.getSheetByName(ALERTS_SHEET_NAME); if (!alerts) { alerts = ss.insertSheet(ALERTS_SHEET_NAME); alerts.appendRow(['Timestamp', 'ItemName', 'Detail']); } alerts.appendRow([new Date(), itemName, detailText]); if (ALERT_EMAILS.length) { MailApp.sendEmail({ to: ALERT_EMAILS.join(','), subject: `Low stock: ${itemName}`, body: `${itemName} is ${detailText}.` }); } }
